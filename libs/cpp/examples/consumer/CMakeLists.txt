cmake_minimum_required(VERSION 3.28)
project(CardanoInstalledConsumer LANGUAGES CXX)

set(CARDANO_COMPONENT "lib" CACHE STRING "XRAYCardanoLib component to consume")
find_package(XRAYCardanoLib CONFIG REQUIRED COMPONENTS "${CARDANO_COMPONENT}")

set(expected_targets core)
if(CARDANO_COMPONENT MATCHES "^(crypto|chain|cip|plutus|lib)$")
  list(APPEND expected_targets crypto)
endif()
if(CARDANO_COMPONENT MATCHES "^(chain|cip|plutus|lib)$")
  list(APPEND expected_targets chain)
endif()
if(CARDANO_COMPONENT MATCHES "^(cip|lib)$")
  list(APPEND expected_targets cip)
endif()
if(CARDANO_COMPONENT MATCHES "^(plutus|lib)$")
  list(APPEND expected_targets plutus)
endif()
if(CARDANO_COMPONENT STREQUAL "lib")
  list(APPEND expected_targets lib)
endif()
foreach(candidate core crypto chain cip plutus lib)
  if(candidate IN_LIST expected_targets)
    if(NOT TARGET "cardano::${candidate}")
      message(FATAL_ERROR "Expected component target cardano::${candidate} is absent")
    endif()
  elseif(TARGET "cardano::${candidate}")
    message(FATAL_ERROR "Unrequested component target cardano::${candidate} is present")
  endif()
endforeach()

add_executable(cardano_installed_consumer main.cpp)
target_compile_features(cardano_installed_consumer PRIVATE cxx_std_23)
target_link_libraries(cardano_installed_consumer PRIVATE "cardano::${CARDANO_COMPONENT}")
string(TOUPPER "${CARDANO_COMPONENT}" component_define)
target_compile_definitions(cardano_installed_consumer PRIVATE "CARDANO_COMPONENT_${component_define}=1")
